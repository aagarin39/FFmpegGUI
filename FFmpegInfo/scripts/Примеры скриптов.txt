@echo off
setlocal enabledelayedexpansion

:: === Параметры путей ===
set "videosource=D:\Торрент"
set "folder=%~dp0bin"
set "videooutput=%videosource%\output"
set "videoconverted=%videosource%\converted"

:: === Проверка переменных ===
rem Рабочая папка: "%folder%"
rem Источник: "%videosource%"
rem Выход: "%videooutput%"
rem Перемещённые: "%videoconverted%"

:: === Переход в рабочую папку ===
cd /d "%videosource%"

:: === Создание выходных директорий ===
md "%videooutput%" 2>nul
md "%videoconverted%" 2>nul

:: === Обработка видеофайлов ===
for %%I in (*.*) do (
    set "channels="

    :: === Получение количества аудиоканалов ===
    for /f "delims=" %%A in (
        'cmd /c ""%folder%\ffprobe" -v error -select_streams a:0 -show_entries stream=channels -of default=noprint_wrappers=1:nokey=1 "%%~fI""'
    ) do (
        set "channels=%%A"
    )

    :: === Очистка и проверка значения каналов ===
    if not defined channels (
        set "channels=2"
    ) else (
        set "channels=!channels: =!"
    )

    :: === Настройка параметров аудио ===
    if "!channels!"=="6" (
        set "abitrate=448k"
        set "achannels=6"
    ) else if "!channels!"=="8" (
        set "abitrate=512k"
        set "achannels=8"
    ) else (
        set "abitrate=192k"
        set "achannels=2"
    )

    :: === Лог обработки ===
    echo --------------------------------------
    echo Обработка: %%~nxI
    echo Аудиоканалов: !channels!, аудио: !achannels! / !abitrate!
    echo Проверка ffmpeg пути: "%folder%\ffmpeg"

    :: === Конвертация видео ===
    "%folder%\ffmpeg" -hwaccel cuda -i "%%~fI" -n ^
        -map 0:0 -c:v h264_nvenc -rc vbr -cq 20 -g 250 -tune hq -rc-lookahead 60 ^
		-vf "format=yuv420p" ^
        -map 0:a -c:a aac -ac !achannels! -b:a !abitrate! ^
        -map 0:s:? -c:s srt ^
        "%videooutput%\%%~nI-264.mkv"

   :: === Перемещение оригинала при успехе ===
    if exist "%videooutput%\%%~nI-264.mkv" (
        move "%%~fI" "%videoconverted%"
    )
)

echo.
echo Готово.
pause
endlocal

	:: === Настройка временного и выходного файла ===
    set "trf_file=%%~nI.trf"
    set "outfile=%%~nI-stab.mkv"

    :: === Первый проход: анализ движения ===
    "%folder%\ffmpeg" -y -i "%%~fI" ^
        -vf vidstabdetect=stepsize=6:shakiness=8:accuracy=9:result=!trf_file! ^

        "%videooutput%\!outfile!"

    :: === Удаление временного файла трека ===
    if exist "!trf_file!" del /f /q "!trf_file!"

        move "%%~fI" "%videoconverted%"
    )
